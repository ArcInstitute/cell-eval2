% Shared preamble for the vcc2026 metric specification.
% The body is generated from vcc2026-metrics.md by build.sh, so everything here has to be
% tolerant of what pandoc emits and of what a Markdown author reasonably types.
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{mathpazo}
\linespread{1.05}
\usepackage{lmodern}
\renewcommand{\rmdefault}{ppl}   % keep Palatino text; lmodern supplies tt/sf
\usepackage{microtype}

\usepackage[a4paper,margin=2.6cm,headheight=14pt]{geometry}
\usepackage{amsmath,amssymb}
\usepackage{booktabs}
\usepackage{array}
\usepackage{longtable}
\usepackage{enumitem}
\usepackage{fancyhdr}
\usepackage{titlesec}
\usepackage{etoolbox}
\usepackage[table]{xcolor}
\usepackage[hidelinks]{hyperref}

% --- structural styling -----------------------------------------------------
\titleformat{\chapter}[display]
  {\normalfont\LARGE\bfseries}{\normalsize\MakeUppercase{\chaptertitlename}\ \thechapter}{8pt}{\LARGE}
\titlespacing*{\chapter}{0pt}{-18pt}{24pt}

\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0.4pt}
\fancyhead[L]{\small\nouppercase{\leftmark}}
\fancyhead[R]{\small\thepage}
\fancypagestyle{plain}{\fancyhf{}\renewcommand{\headrulewidth}{0pt}\fancyfoot[C]{\small\thepage}}

% Markdown carries no line-breaking hints, and the metric names are long unbreakable runs;
% give the justifier some slack rather than let a paragraph run into the margin.
\setlength{\emergencystretch}{3em}

% Block paragraphs. The source is a sequence of bold-led topic paragraphs, which an indent
% alone does not separate strongly enough once they are set solid.
\setlength{\parindent}{0pt}
\setlength{\parskip}{7pt plus 2pt minus 1pt}

\setlist[itemize]{topsep=3pt,itemsep=2pt,parsep=0pt,leftmargin=1.4em}
\setlist[enumerate]{topsep=3pt,itemsep=2pt,parsep=0pt,leftmargin=1.6em}

\renewcommand{\arraystretch}{1.15}
\setlength{\LTpre}{8pt}
\setlength{\LTpost}{8pt}
\newcolumntype{L}[1]{>{\raggedright\arraybackslash}p{#1}}

% Markdown emits no per-table sizing, and the metric names are long; shrink every longtable
% (which is what pandoc produces) rather than annotate the source.
\AtBeginEnvironment{longtable}{\footnotesize}

% --- what pandoc expects to exist -------------------------------------------
\providecommand{\tightlist}{\setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
\providecommand{\passthrough}[1]{#1}

% Monospaced identifiers. Underscores get a break opportunity so that long metric names wrap
% inside a narrow table column instead of running past the margin. Applied to \texttt itself
% because that is what pandoc emits for Markdown `code` spans.
% Math-mode safe: \ttfamily is illegal there, and a Markdown author can legitimately put a
% `code` span inside a formula.
\let\origtexttt\texttt
\renewcommand{\texttt}[1]{%
  \ifmmode\text{\origtexttt{#1}}\else{\ttfamily\def\_{\char`\_\allowbreak}#1}\fi}
\newcommand{\code}[1]{\texttt{#1}}
